cmake_minimum_required(VERSION 3.20.2)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(gameframe_kcp VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(GFNET_MBEDTLS_SOURCE "" CACHE PATH "Verified prepared mbedTLS release")
set(GFNET_KCP_SOURCE "" CACHE PATH "Hash-verified prepared KCP sources")
option(GFNET_TESTS "Build host-only native regression harness" OFF)
option(GFNET_SANITIZERS "Enable ASan and UBSan for host regression" OFF)
if(NOT EXISTS "${GFNET_MBEDTLS_SOURCE}/gfnet-source-lock.json" OR NOT EXISTS "${GFNET_KCP_SOURCE}/kcp-source-manifest.json")
    message(FATAL_ERROR "Run explicit dependency preparation before building")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/mbedtls_config.h" CACHE FILEPATH "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/crypto_config.h" CACHE FILEPATH "" FORCE)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/config")
if(GFNET_SANITIZERS)
    if(MSVC OR ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
        message(FATAL_ERROR "Sanitizer builds are host-only and cannot be installed")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory("${GFNET_MBEDTLS_SOURCE}" "mbedtls" EXCLUDE_FROM_ALL)
set(GFNET_SOURCES src/gfnet_memory.c src/gfnet_connection.c src/gfnet_listener.c "${GFNET_KCP_SOURCE}/ikcp.c")
add_library(gfnet_core STATIC ${GFNET_SOURCES})
target_include_directories(gfnet_core PUBLIC include PRIVATE src "${GFNET_KCP_SOURCE}")
target_link_libraries(gfnet_core PUBLIC mbedtls)
if(WIN32)
    target_link_libraries(gfnet_core PUBLIC bcrypt)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(gfnet_core PUBLIC Threads::Threads)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_library(gameframe_kcp STATIC ${GFNET_SOURCES})
else()
    add_library(gameframe_kcp SHARED ${GFNET_SOURCES})
endif()
target_include_directories(gameframe_kcp PUBLIC include PRIVATE src "${GFNET_KCP_SOURCE}")
target_link_libraries(gameframe_kcp PRIVATE mbedtls)
if(WIN32)
    set_target_properties(gameframe_kcp PROPERTIES PREFIX "")
    target_link_libraries(gameframe_kcp PRIVATE bcrypt)
else()
    target_link_libraries(gameframe_kcp PRIVATE Threads::Threads)
endif()
if(GFNET_TESTS)
    target_compile_definitions(gfnet_core PRIVATE GFNET_TESTING)
    enable_testing()
    add_executable(gfnet_regression tests/gfnet_regression.c)
    target_include_directories(gfnet_regression PRIVATE src "${GFNET_KCP_SOURCE}")
    target_link_libraries(gfnet_regression PRIVATE gfnet_core)
    target_compile_definitions(gfnet_regression PRIVATE GFNET_TESTING)
    add_test(NAME gfnet_regression COMMAND gfnet_regression)
endif()
